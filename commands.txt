# sanity check on the default L4
uv run modal run modal_run.py --cmd "nvidia-smi && nsys --version"

# one benchmark run (small model, full train step)
uv run modal run modal_run.py --gpu H100 --cmd "python -m cs336_systems.benchmarking_script --device cuda --d_model 768 --d_ff 3072 --num_layers 12 --num_heads 12 --mode train"

# all model sizes x all modes in one container
uv run modal run modal_run.py --gpu H100 --timeout 1800 --cmd '
for size in "small 768 3072 12 12" "medium 1024 4096 24 16" "large 1280 5120 36 20" "xl 2560 10240 32 32" "10B 4608 12288 50 36"; do
  set -- $size
  for mode in forward backward train; do
    echo "== $1 $mode"
    python -m cs336_systems.benchmarking_script --device cuda --d_model $2 --d_ff $3 --num_layers $4 --num_heads $5 --mode $mode 2>&1 | grep -E "parameters|mean|std|OutOfMemory" || echo "failed"
  done
done'

# local check on CPU with a tiny model
uv run python -m cs336_systems.benchmarking_script --device cpu --d_model 64 --num_layers 2 --num_heads 4 --d_ff 128 --context_length 32 --mode train

# nsys profile (small model, train mode) and download the report
uv run modal run modal_run.py --cmd "mkdir -p runs/nsys && nsys profile --trace=cuda,nvtx,osrt --pytorch=autograd-nvtx --force-overwrite=true -o runs/nsys/small_train python -m cs336_systems.nsys_profile --device cuda --d_model 768 --d_ff 3072 --num_layers 12 --num_heads 12 --mode train --warmup_steps 2 --num_steps 3"
uv run modal volume get --force cs336-a2-runs nsys/small_train.nsys-rep runs/nsys/small_train.nsys-rep

# gradient checkpointing sweep (xl, ctx 2048, forward + backward), 0 = no checkpointing
uv run modal run modal_run.py --gpu H100 --timeout 1200 --cmd '
for k in 1 2 4 8 0; do
  echo "== xl ctx=2048 backward checkpoint_every=$k"
  python -m cs336_systems.benchmarking_script --device cuda --d_model 2560 --d_ff 10240 --num_layers 32 --num_heads 32 --context_length 2048 --batch_size 4 --mode backward --warmup_steps 2 --num_steps 3 --checkpoint_every $k 2>&1 | grep -E "parameters|mean time|peak memory|OutOfMemory" || echo "failed"
done'

# attention alone: sequence length x head size sweep
uv run modal run modal_run.py --gpu H100 --timeout 1200 --cmd "python -m cs336_systems.benchmark_attention"
# same sweep, uncompiled and torch.compile side by side
uv run modal run modal_run.py --gpu H100 --timeout 1800 --cmd "python -m cs336_systems.benchmark_attention --compile"

# whole model, uncompiled vs torch.compile, all sizes and modes
uv run modal run modal_run.py --gpu H100 --timeout 3600 --cmd '
for size in "small 768 3072 12 12" "medium 1024 4096 24 16" "large 1280 5120 36 20" "xl 2560 10240 32 32"; do
  set -- $size
  for mode in forward backward train; do
    for c in "" "--compile"; do
      echo "== $1 $mode ${c:-eager}"
      python -m cs336_systems.benchmarking_script --device cuda --d_model $2 --d_ff $3 --num_layers $4 --num_heads $5 --mode $mode $c 2>&1 | grep -E "mean time|std dev|OutOfMemory" || echo "failed"
    done
  done
done'
